## Usage

### Basic Link

  <p>
    The default Link component renders an underlined anchor with primary color styling.
  </p>

```tsx
import { Link } from "@cloudflare/kumo";

export default function Example() {
  return (
    <p>
      Read our <Link href="/docs">documentation</Link> for more details.
    </p>
  );
}
```

### Opening links in a new tab

<p>
  Set `target="_blank"` to open a link in a new tab. Kumo adds
  `rel="noopener noreferrer"`, a visible icon, and a screen-reader announcement
  such as "Visit Cloudflare (opens in a new tab)". Don't add the icon or the
  announcement yourself.
</p>

```tsx
<Link href="https://cloudflare.com" target="_blank">
  Visit Cloudflare
</Link>
```

### Framework Integration (LinkProvider)

<p>
  Configure a `LinkProvider` once at your app root. Call sites then only need
  an `href`. Kumo resolves each link before rendering: internal routes render
  your router's link component, and everything else (other origins, `mailto:`,
  `tel:`, downloads, and in-page fragments like `#section`) renders a plain
  `&lt;a&gt;`.
</p>

```tsx
import { forwardRef } from "react";
import { LinkProvider, type RouterLinkProps } from "@cloudflare/kumo";
import { Link as ReactRouterLink } from "react-router-dom";

// Receives internal links only. Map href to the router's destination prop.
const RouterLink = forwardRef<HTMLAnchorElement, RouterLinkProps>(
  ({ href, ...props }, ref) => (
    <ReactRouterLink ref={ref} to={href} {...props} />
  ),
);
RouterLink.displayName = "RouterLink";

export function App() {
  return (
    <LinkProvider router={RouterLink}>
      <YourApp />
    </LinkProvider>
  );
}
```

<p>
  The provider also holds app-wide navigation policy:
</p>

```tsx
// Paths owned by another app on the same origin
const PEER_APP_PATH = /^\/(?:[^/]+\/)?one(?:\/|$)/;
const isClientRoute = (url: URL) => !PEER_APP_PATH.test(url.pathname);

<LinkProvider
  router={RouterLink}
  externalLinks="new-tab"
  isClientRoute={isClientRoute}
  origin="https://dash.cloudflare.com"
>
  <App />
</LinkProvider>;
```

<ul class="ml-4 list-disc space-y-1">
  <li>
    `externalLinks`: `"same-tab"` (default) or `"new-tab"` for links to other
    origins. Opening new tabs can be disorienting, so only opt in when your app
    needs it.
  </li>
  <li>
    `isClientRoute`: return `false` to send an internal URL to a plain
    `&lt;a&gt;` instead of the router. Define it outside the component (or
    with `useCallback`) so it stays stable between renders.
  </li>
  <li>
    `origin`: lets Kumo treat absolute URLs on your own origin as internal
    routes. In server-rendered apps, pass the same value on the server and the
    client; don't read `window.location` while rendering.
  </li>
</ul>

<p>
  A link's own `target` always wins. `target="_blank"` opens any link in a new
  tab, including an internal path, and `target="_self"` keeps it in the current
  tab even when the provider opens external links in new tabs. Links with a
  `target` other than `_self` don't use client-side routing.
</p>

### Custom link UI (useLink)

<p>
  For cards or other custom links, use `useLink`. It returns the element to
  render, the resolved anchor props, and the new-tab cue. Render all three.
</p>

```tsx
import { useLink } from "@cloudflare/kumo";

function ZoneCard({ href, name }: { href: string; name: string }) {
  const { Component, linkProps, newTabCue } = useLink({ href });
  return (
    <Component {...linkProps} className="block rounded-lg bg-kumo-elevated p-4">
      {name}
      {newTabCue}
    </Component>
  );
}
```

<ComponentSection>

## Design Guidelines

  <div class="space-y-4 text-sm">
    <div>

      ### When to Use Each Variant

      <ul class="ml-4 list-disc space-y-1">
        <li>
          <strong>inline</strong>: Default choice for links within body text
        </li>
        <li>
          <strong>current</strong>: Links inside alerts, banners, or other
          colored containers
        </li>
        <li>
          <strong>plain</strong>: Navigation menus, footers, or where underlines
          are distracting
        </li>
      </ul>
    </div>
    <div>

      ### New Tabs

      <ul class="ml-4 list-disc space-y-1">
        <li>Prefer the current tab. New tabs can be disorienting.</li>
        <li>
          Set new-tab policy once with `externalLinks` on `LinkProvider`, not
          per link
        </li>
        <li>
          Don't add icons, `rel`, or "(opens in a new tab)" text yourself; Kumo
          adds them
        </li>
      </ul>
    </div>
    <div>

      ### Framework Integration

      <ul class="ml-4 list-disc space-y-1">
        <li>
          Pass your router's link component to `LinkProvider` as `router`
        </li>
        <li>
          The router component only receives internal links; map `href` to its
          destination prop (e.g. React Router's `to`)
        </li>
        <li>
          Use `useLink` for custom link UI such as cards
        </li>
      </ul>
    </div>
    <div>

      ### Accessibility

      <ul class="ml-4 list-disc space-y-1">
        <li>Links are keyboard focusable by default</li>
        <li>
          Links that open in a new tab announce it, whether they're named by
          their content, `aria-label`, or `aria-labelledby`
        </li>
        <li>
          Translate the announcement with `KumoLocaleProvider`
          (`link.opensInNewTab`)
        </li>
        <li>Ensure sufficient color contrast for all variants</li>
        <li>Use descriptive link text (avoid "click here")</li>
      </ul>
    </div>
  </div>
